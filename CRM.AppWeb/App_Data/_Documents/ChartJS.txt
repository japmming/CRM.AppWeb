$$ChartJsBar('ID_DE_CHART_JS').create({
    title: 'Titulo de Prueba',													/*Titulo (11/09/2020 | aun sin funcion)*/
    headers: ['Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio'],			/*Nombres de el Eje X*/
    data: ['10¦15¦20¦25¬30¦35¦40¬50¦55¦60¬70¦75¦80¬90¦95¦100¬90¦85¦80'],		/*Data (La canditada de split(¬) debe ser igual a la "header" pero no es obligatorio en split(¦))*/

    NTitleY: 10,																/*Cantidad de lineas de cuadricula Y*/
    heightY: 25,																/*Espacio entre las lineas de cuadricula Y (px)*/
    showGridY: true,															/*¿Desea mostrar las lineas de cuadricula?*/
    separatorBar :'¦',															/*separador de data dentro de un header (mes)*/
    separatorHead: '¬',															/*separador de LISTAS PARA CADA HEADER (mes)*/
    bgColors: ['#00b6de¦#7ac143¦#ff8200¦#f05587¦#22bcb9'],						/*Colores para las barras (de haber mas colores que barras se usara hasta el [i] necesario)*/
    ColorsClass: false,															/*Alternar entre color por "style" o por "class"*/
    bgColorsClass:[],															/*Colores para las barras en Clase*/
    namespace: namespace														/*Namespace*/
});
